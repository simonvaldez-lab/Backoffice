        function renderizarANS(matriz) {
            var ans = matriz || (typeof obtenerConfigANS === 'function' ? obtenerConfigANS() : []);
            var tbody = document.getElementById('tbody-ans');
            if (!tbody) return;
            tbody.innerHTML = '';

        function editarANS(idx) {
            var ans = typeof obtenerConfigANS === 'function' ? obtenerConfigANS() : [];
            var item = ans[idx];
            if (!item) return;

            var nuevoMontaje = prompt("⏱️ TIEMPO MÁXIMO DE MONTAJE (PREPARADOR):\n\nIngresa el límite en minutos para " + item.nombre + ":", item.maxMontajeMin);
            if (nuevoMontaje !== null && !isNaN(nuevoMontaje) && Number(nuevoMontaje) > 0) {
                var nuevoAprob = prompt("⏱️ TIEMPO MÁXIMO DE APROBACIÓN (KATE):\n\nIngresa el límite en minutos para el check final:", item.maxAprobacionMin);
                if (nuevoAprob !== null && !isNaN(nuevoAprob) && Number(nuevoAprob) > 0) {
                    item.maxMontajeMin = Number(nuevoMontaje);
                    item.maxAprobacionMin = Number(nuevoAprob);
                    
                    localStorage.setItem("bold_config_ans", JSON.stringify(ans));
                    renderizarANS(ans);

                    getDb().collection("configuracion").doc("ans").set({
                        lista: ans
                    }).then(function() {
                        alert("✅ ¡Matriz de ANS actualizada y guardada exitosamente en Google Cloud!");
                    }).catch(function(err) {
                        alert("⚠️ Guardado en pantalla pero falló la nube: " + err.message);
                    });
                }
            }
        }

        function activarEscuchadorANSRealTime() {
            try {
                getDb().collection("configuracion").doc("ans").onSnapshot(function(doc) {
                    if (doc.exists && doc.data().lista) {
                        var matrizNube = doc.data().lista;
                        localStorage.setItem("bold_config_ans", JSON.stringify(matrizNube));
                        renderizarANS(matrizNube);
                        console.log("⚡ [WEBSOCKET ADMIN]: Matriz ANS sincronizada desde Firestore en vivo.");
                    }
                });
            } catch(e) { console.error("Error activando WebSocket ANS:", e); }
        }

        window.addEventListener('DOMContentLoaded', function() {
            renderizarUsuarios();
            renderizarANS();
            activarEscuchadorANSRealTime();
        });

const fs = require('fs');
const path = require('path');

// Solo leeremos estos tipos de archivos para no saturar con imágenes o binarios
const extensionesPermitidas = ['.html', '.js', '.css'];
// Carpetas que NO queremos leer
const carpetasIgnoradas = ['node_modules', '.git', '.github'];
// El archivo donde guardaremos todo
const archivoSalida = 'codigo_completo.txt';

let contenidoTotal = '======================================================\n';
contenidoTotal += '🚀 REPORTE COMPLETO DEL CÓDIGO - BOLD TREASURY\n';
contenidoTotal += '======================================================\n';

function leerDirectorio(directorio) {
    const archivos = fs.readdirSync(directorio);
    
    archivos.forEach(archivo => {
        const rutaCompleta = path.join(directorio, archivo);
        const stats = fs.statSync(rutaCompleta);
        
        if (stats.isDirectory()) {
            if (!carpetasIgnoradas.includes(archivo)) {
                leerDirectorio(rutaCompleta); // Leer subcarpetas (como js/)
            }
        } else {
            const ext = path.extname(archivo);
            if (extensionesPermitidas.includes(ext)) {
                const contenido = fs.readFileSync(rutaCompleta, 'utf8');
                contenidoTotal += `\n\n\n//////////////////////////////////////////////////////\n`;
                contenidoTotal += `📁 ARCHIVO: ${rutaCompleta}\n`;
                contenidoTotal += `//////////////////////////////////////////////////////\n\n`;
                contenidoTotal += contenido;
            }
        }
    });
}

try {
    // Iniciar lectura desde la carpeta actual
    leerDirectorio('./');
    fs.writeFileSync(archivoSalida, contenidoTotal);
    console.log(`✅ ¡ÉXITO! Se unió todo tu código en el archivo -> ${archivoSalida}`);
} catch (e) {
    console.error(`❌ Error al consolidar: ${e.message}`);
}

const { getFirestore, FieldValue, Timestamp } = require("firebase-admin/firestore");
const cors = require("cors")({ origin: true });

admin.initializeApp();
const db = getFirestore("treasurybackoffice");

// 🛡️ GUARDIA DE SEGURIDAD (RBAC - ZERO TRUST)
async function validarPermiso(usuarioCliente, rolesPermitidos) {
    if (!usuarioCliente || !usuarioCliente.correo) throw new Error("Acceso Denegado: Sesión no identificada.");
    const correo = usuarioCliente.correo.toLowerCase().trim();
    if (correo === "simon.valdez@bold.co") return { autorizado: true, rol: "maestro", nombre: "Simon Valdez (Director)" };

    const docRef = await db.collection("configuracion").doc("usuarios").get();
    let listaUsuarios = docRef.exists && docRef.data().lista ? docRef.data().lista : [
        { correo: "lau@bold.co", rol: "solicitante", estado: "Activo" },
        { correo: "fel@bold.co", rol: "preparador", estado: "Activo" },
        { correo: "kat@bold.co", rol: "aprobador", estado: "Activo" }
    ];

    const usr = listaUsuarios.find(u => u.correo.toLowerCase() === correo);
    if (!usr) throw new Error(`Acceso Denegado: El correo ${correo} no está autorizado.`);
    if (usr.estado !== "Activo") throw new Error(`Acceso Bloqueado: Usuario suspendido.`);
    
    if (usr.rol === "maestro" || rolesPermitidos.includes(usr.rol)) {
        return { autorizado: true, rol: usr.rol, nombre: usr.nombre || correo };
    } else {
        throw new Error(`Violación RBAC: Tu rol actual ('${usr.rol.toUpperCase()}') no tiene permisos para este paso.`);
    }
}

// ⏱️ CÁLCULO DE VENCIMIENTO ANS
async function calcularVencimientoANS(prioridad, esAprobacion = false) {
    const docRef = await db.collection("configuracion").doc("ans").get();
    let matriz = docRef.exists && docRef.data().lista ? docRef.data().lista : [
        { prioridad: 1, maxMontajeMin: 15, maxAprobacionMin: 10 },
        { prioridad: 2, maxMontajeMin: 30, maxAprobacionMin: 20 },
        { prioridad: 3, maxMontajeMin: 60, maxAprobacionMin: 45 }
    ];
    const prioNum = Number(prioridad) || 2;
    const regla = matriz.find(m => m.prioridad === prioNum) || matriz[1];
    const minutos = esAprobacion ? regla.maxAprobacionMin : regla.maxMontajeMin;
    const ahora = new Date();
    ahora.setMinutes(ahora.getMinutes() + minutos);
    return Timestamp.fromDate(ahora);
}

function crearRegistroHistorial(paso, detalle, usr, alerta = false) {
    const ahora = new Date();
    const militar = String(ahora.getHours()).padStart(2, '0') + ":" + String(ahora.getMinutes()).padStart(2, '0');
    return {
        fecha: ahora.toLocaleDateString('es-CO') + " " + militar,
        paso: paso,
        detalle: `${detalle} | Operado por: ${usr.nombre || usr.correo} (${(usr.rol || '').toUpperCase()})`,
        alerta: alerta
    };
}

exports.apiGuardarUsuarios = functions.https.onRequest((req, res) => {
    return cors(req, res, async () => {
        try {
            if (req.method !== "POST") return res.status(405).json({ exito: false });
            await validarPermiso(req.body.usuario, ["maestro"]);
            await db.collection("configuracion").doc("usuarios").set({ lista: req.body.listaUsuarios, actualizadoEn: FieldValue.serverTimestamp() });
            return res.status(200).json({ exito: true });
        } catch (err) { return res.status(403).json({ exito: false, error: err.message }); }
    });
});

exports.apiRadicarOperacion = functions.https.onRequest((req, res) => {
    return cors(req, res, async () => {
        try {
            if (req.method !== "POST") return res.status(405).json({ exito: false });
            const datos = req.body;
            const usrValidado = await validarPermiso(datos.usuario, ["solicitante"]);
            if (!datos.radicado || !datos.montoSol) return res.status(400).json({ exito: false, error: "Datos incompletos." });

            const fechaVencimiento = await calcularVencimientoANS(datos.prioridad, false);
            const operacionDB = {
                ...datos,
                estado: "Pendiente Validación", // Estado inicial para que Felipe haga el primer check
                fechaServidor: FieldValue.serverTimestamp(),
                fechaVencimientoSLA: fechaVencimiento,
                creadoEnCloud: true,
                historial: [crearRegistroHistorial("1. 🟢 RADICACIÓN INICIAL", "Solicitud creada e ingresada a bandeja de Preparador para Check KYC.", usrValidado)]
            };

            await db.collection("operaciones").doc(datos.radicado).set(operacionDB);
            console.log(`✅ [CLOUD] Radicado: ${datos.radicado}`);
            return res.status(200).json({ exito: true, id: datos.radicado });
        } catch (err) { return res.status(403).json({ exito: false, error: err.message }); }
    });
});

// 🔥 NUEVO ENDPOINT: PASO 1 (VALIDACIÓN KYC ABSORBIDA POR PREPARADOR)
exports.apiValidarKYC = functions.https.onRequest((req, res) => {
    return cors(req, res, async () => {
        try {
            if (req.method !== "POST") return res.status(405).json({ exito: false });
            const { radicado, usuario } = req.body;
            
            // 🛡️ RBAC: Ahora el PREPARADOR está autorizado para ejecutar la validación inicial
            const usrValidado = await validarPermiso(usuario, ["preparador"]);
            if (!radicado) return res.status(400).json({ exito: false, error: "Falta radicado." });

            const docRef = db.collection("operaciones").doc(radicado);
            const doc = await docRef.get();
            if (!doc.exists) return res.status(404).json({ exito: false, error: "Operación no encontrada." });

            const nuevoHistorial = crearRegistroHistorial("2. 🛡️ CHECK KYC / COMPLIANCE APROBADO", "Revisión antifraude KYC certificada por el Preparador. Lista para iniciar montaje en portal bancario.", usrValidado);

            await docRef.update({
                estado: "En Preparación",
                kycValidadoPor: usrValidado.nombre,
                historial: FieldValue.arrayUnion(nuevoHistorial),
                actualizadoEn: FieldValue.serverTimestamp()
            });

            console.log(`🛡️ [CLOUD] KYC Validado por Preparador: ${radicado}`);
            return res.status(200).json({ exito: true, mensaje: "KYC certificado. Operación habilitada para montaje bancario." });
        } catch (err) { return res.status(403).json({ exito: false, error: err.message }); }
    });
});

function obtenerFechaHoraMilitar() {
    const d = new Date();
    const dia = String(d.getDate()).padStart(2, "0");
    const mes = String(d.getMonth() + 1).padStart(2, "0");
    const anio = d.getFullYear();
    const hrs = String(d.getHours()).padStart(2, "0");
    const min = String(d.getMinutes()).padStart(2, "0");
    return `${dia}/${mes}/${anio} - ${hrs}:${min}`;
}

function obtenerFechaISO() {
    const d = new Date();
    const dia = String(d.getDate()).padStart(2, "0");
    const mes = String(d.getMonth() + 1).padStart(2, "0");
    const anio = d.getFullYear();
    return `${anio}-${mes}-${dia}`;
}

function fechasCoinciden(fechaOp, fechaInput) {
    if (!fechaInput || fechaInput === "") return true;
    if (!fechaOp) return false;
    const limpiaOp = String(fechaOp).split(" - ")[0].trim();
    const partes = limpiaOp.split("/");
    if (partes.length !== 3) return false;
    const fNorm = `${partes[2]}-${String(partes[1]).padStart(2, "0")}-${String(partes[0]).padStart(2, "0")}`;
    return fNorm === fechaInput;
}

        document.querySelectorAll('.menu-item').forEach(item => {

        });

            while (next && !next.classList.contains('menu-categoria')) {

                next = next.nextElementSibling;
            }
            if (!visible) cat.style.display = 'none';
        });

// === MÓDULO DE CICLO DE VIDA, GUARDIAS RBAC Y RECHAZO ESTRICTO ===
// 🛡️ GUARDIA RBAC: Verifica que el rol sea autorizado para rechazar o aprobar
function guardiaRolRechazo(usr) {
    var rol = String(usr.rol || "").toLowerCase().trim();
    if (rol === "solicitante") {
        alert("⛔ ACCESO DENEGADO (RBAC):" + "\n" + "El rol Solicitante solo puede iniciar solicitudes y monitorear su estado. No tiene permisos para rechazar, modificar ni aprobar operaciones.");
        throw new Error("Violación RBAC: Solicitante intentando rechazar.");
    }
    if (rol !== "preparador" && rol !== "aprobador" && rol !== "maestro") {
        alert("⛔ ACCESO DENEGADO (RBAC):" + "\n" + "Tu rol no cuenta con privilegios para modificar el ciclo de vida de este radicado.");
        throw new Error("Violación RBAC: Rol no autorizado.");
    }
    return true;
}

function ejecutarRechazoSeguro(radicado) {
    try {
        var usrRaw = sessionStorage.getItem("usuarioLogueado") || localStorage.getItem("bold_ultimo_usuario_backup") || "{}";
        var usr = JSON.parse(usrRaw);
        guardiaRolRechazo(usr);
        var ops = typeof obtenerOperaciones === "function" ? obtenerOperaciones() : [];
        var op = ops.find(function(o) { return o.radicado === radicado; });
        if (!op) return alert("❌ No se encontró el radicado especificado: " + radicado);
        var motivo = prompt("🚨 MOTIVO DE DEVOLUCIÓN / RECHAZO (OBLIGATORIO):" + "\n\n" + "Por favor describe detalladamente el error encontrado para que el Solicitante (" + (op.solicitante || "Analista") + ") pueda corregirlo y volver a radicar:");
        if (motivo === null) return;
        if (!motivo || motivo.trim() === "") {
            alert("⚠️ VALIDACIÓN FALLIDA: Es absolutamente obligatorio escribir las observaciones o motivo de rechazo. No se puede devolver una operación en blanco.");
            return ejecutarRechazoSeguro(radicado);
        }
        var motivoLimpio = motivo.trim();
        var rechazador = (usr.nombre || "Analista") + " (" + (usr.rol || "Preparador").toUpperCase() + ")";
        op.estado = "RECHAZADO";
        op.motivoRechazo = motivoLimpio;
        op.rechazadoPor = rechazador;
        op.enTransito = false;
        if (!op.historial) op.historial = [];
        op.historial.push({
            fecha: typeof obtenerFechaHoraMilitar === "function" ? obtenerFechaHoraMilitar() : new Date().toLocaleString(),
            paso: "⚫ OPERACIÓN RECHAZADA / DEVUELTA",
            detalle: "Devuelto por " + rechazador + ". Motivo obligatorio registrado: \"" + motivoLimpio + "\"",
            alerta: true
        });

function enviarRadicadoACloudFunction(operacion) {
    try {
        if (!operacion || !operacion.radicado) return;
        
        // Evitamos reenviar si es un guardado de actualización o rechazo posterior
        if (operacion.estado !== "Pendiente Validación" && operacion.estado !== "Pendiente Preparación") return;

        console.log("☁️ [PUENTE CLOUD] Enviando radicado a Firebase Cloud Functions: " + operacion.radicado);

        // Petición asíncrona en segundo plano (Fetch API) para no congelar la UI de Laura
        fetch(CLOUD_FUNCTION_URL, {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
                "Accept": "application/json"
            },
            body: JSON.stringify(operacion)
        })
        .then(function(res) { return res.json(); })
        .then(function(data) {
            if (data.exito) {
                console.log("🟢 [CLOUD FUNCTION OK]: Operación asegurada en la nube -> " + data.idDocumento);
            } else {
                console.warn("⚠️ [CLOUD FUNCTION RESPUESTA]: " + (data.error || "Desconocido"));
            }
        })
        .catch(function(err) {
            // Si la Cloud Function aún no está desplegada o hay fallo de red, el sistema local sigue intacto
            console.warn("ℹ️ [MODO OFFLINE / LOCAL]: No se pudo contactar la Cloud Function aún. La operación permanece asegurada en la memoria local y se sincronizará luego. Detalle: " + err.message);
        });
    } catch(e) {
        console.error("Fallo silencioso en puente Cloud:", e.message);
    }
}

// Función para calcular si una operación está vencida según la configuración real
function auditarCumplimientoANS(operacion) {
    if (!operacion || !operacion.fechaRadicacion) return { vencido: false, minTranscurridos: 0, limiteMin: 30 };
    var config = obtenerConfigANS();
    var prio = operacion.prioridad || 2;
    var regla = config.find(function(c) { return c.prioridad === prio; }) || config[1];
    
    // Si está en preparación, medimos contra maxMontajeMin; si está en aprobación, contra maxAprobacionMin
    var limite = (operacion.estado === "En Aprobación") ? regla.maxAprobacionMin : regla.maxMontajeMin;
    return { vencido: false, limiteMin: limite, regla: regla };
}

        // 🚀 SENSOR DINÁMICO: DETECTA CUANDO UNA RUTA ADMITE O REQUIERE PLANTILLA EXCEL
        function evaluarReglasExcel() {
            const cat = document.getElementById('niv-cat').value;
            const sub = document.getElementById('niv-sub').value;
            const det = document.getElementById('niv-det').value;
            const panel = document.getElementById('panel-excel-dinamico');
            const contenido = document.getElementById('contenido-excel-dinamico');
            
            if (!panel || !contenido) return;

            let modo = null, icono = "", desc = "";

            // Regla 1: Pago PSE / Pago Proveedores / Pago Proveedores Locales
            if (cat === "Pago PSE" && sub === "Pago Proveedores" && det === "Pago Proveedores Locales") {
                modo = "Dispersión Masiva PSE"; icono = "🔄"; desc = "Lote masivo de transferencias por PSE (Pago Proveedores Locales en Excel .xlsx)";
            } 
            // Regla 2: Pago PSE / Pago Impuestos / ...
            else if (cat === "Pago PSE" && sub === "Pago Impuestos") {
                modo = "Pago de Impuestos (Taxes)"; icono = "🏛️"; desc = "Liquidación y Pago de Impuestos DIAN / Distritales en Excel (.xlsx)";
            }
            // Regla 3: Pago PSE / Payroll / Pacto Colectivo
            else if (cat === "Pago PSE" && sub === "Payroll" && det === "Pacto Colectivo") {
                modo = "Pactos Bancarios"; icono = "📈"; desc = "Inversión / Pacto Colectivo de Recompra en Excel (.xlsx)";
            }
            // Compatibilidad con categorías anteriores
            else if (cat === "Dispersión Masiva") {
                modo = "Dispersión Masiva PSE"; icono = "🔄"; desc = "Dispersión Masiva por Lote ACH (.xlsx)";
            }
            else if (cat === "Inversiones y Pactos") {
                modo = "Pactos Bancarios"; icono = "📈"; desc = "Inversión y Recompras en Excel (.xlsx)";
            }
            else if (cat === "Impuestos y Taxes") {
                modo = "Pago de Impuestos (Taxes)"; icono = "🏛️"; desc = "Impuestos DIAN en Excel (.xlsx)";
            }

            if (modo) {
                contenido.innerHTML = `
                    <div style="background: #F0FDF4; border: 2px dashed #22C55E; padding: 14px 18px; border-radius: 10px; display: flex; justify-content: space-between; align-items: center; animation: fadeIn 0.3s ease-in-out;">
                        <div style="display: flex; align-items: center; gap: 12px;">
                            <span style="font-size: 28px;">${icono}</span>
                            <div>
                                <strong style="color: #166534; font-size: 13px; display: block;">✨ Formato Masivo Detectado: Plantilla ${modo}</strong>
                                <span style="color: #15803D; font-size: 11px;">La ruta operativa seleccionada admite o requiere adjuntar el detalle en formato Excel (.xlsx).</span>
                            </div>
                        </div>
                        <button type="button" class="btn" style="background: #166534; color: white; font-weight: bold; font-size: 12px; padding: 8px 16px; box-shadow: 0 4px 10px rgba(22,101,52,0.2); white-space: nowrap;" onclick="activarModoExcel('${modo}', '${desc}')">
                            ${icono} Adjuntar Plantilla Excel
                        </button>
                    </div>
                `;
                panel.style.display = 'block';
            } else {
                panel.style.display = 'none';
                contenido.innerHTML = '';
            }
        }

        // ACCIÓN AL HACER CLIC EN EL BOTÓN DINÁMICO DE EXCEL
        function activarModoExcel(tipo, descripcion) {
            const nomActual = document.getElementById('nombre-op').value.trim();
            if (!nomActual.startsWith("[Excel]")) {
                document.getElementById('nombre-op').value = `[Excel] ${tipo} - ${new Date().toLocaleDateString('es-CO')}`;
            }
            document.getElementById('doc-instrucciones').value = `Operación basada en plantilla Excel institucional: ${descripcion}. Por favor verificar registros y soportes en el archivo adjunto.`;
            
            const caja = document.getElementById('caja-docs-opcional');
            const flecha = document.getElementById('flecha-docs');
            if (caja && caja.style.display === "none") {
                caja.style.display = "block";
                flecha.innerText = "▲";
            }
            
            const fileInput = document.getElementById('doc-archivo');
            if (fileInput) {
                fileInput.style.border = "2px solid #22C55E";
                fileInput.style.background = "#F0FDF4";
                setTimeout(() => {
                    fileInput.style.border = "1px solid #CBD5E1";
                    fileInput.style.background = "white";
                }, 3500);
            }
            
            alert(`📊 Modo Excel Activado: ${tipo}\n\n1. Se etiquetó el nombre de la operación.\n2. Se abrió la sección inferior y parpadea en verde el campo para que subas tu archivo .XLSX o .XLSM.`);
        }

        function alternarSeccionDocs() {
            const caja = document.getElementById('caja-docs-opcional');
            const flecha = document.getElementById('flecha-docs');
            if (caja.style.display === "none") {
                caja.style.display = "block";
                flecha.innerText = "▲";
            } else {
                caja.style.display = "none";
                flecha.innerText = "▼";
            }
        }

        function ajustarANS() {
            const prio = document.getElementById('prioridad-op').value;
            document.getElementById('ans-op').value = prio === "1" ? "1 Hora" : prio === "2" ? "2 Horas" : "4 Horas";
        }

        function inicializarArbol() {
            const catSelect = document.getElementById('niv-cat');
            catSelect.innerHTML = '<option value="">-- Seleccione Categoría --</option>';
            Object.keys(arbolOperaciones).forEach(c => catSelect.innerHTML += `<option value="${c}">${c}</option>`);
            catSelect.onchange = () => { poblarSubtipos(catSelect.value); evaluarReglasExcel(); };
        }
        function poblarSubtipos(cat) {
            const el = document.getElementById('niv-sub'); el.innerHTML = '<option value="">-- Seleccione Subtipo --</option>'; el.disabled = !cat;
            if(cat && arbolOperaciones[cat]) Object.keys(arbolOperaciones[cat]).forEach(s => el.innerHTML += `<option value="${s}">${s}</option>`);
            el.onchange = () => { poblarDetalles(cat, el.value); evaluarReglasExcel(); };
            limpiarNivelesDesde(3);
            evaluarReglasExcel();
        }
        function poblarDetalles(cat, sub) {
            const el = document.getElementById('niv-det'); el.innerHTML = '<option value="">-- Seleccione Detalle --</option>'; el.disabled = !sub;
            if(sub && arbolOperaciones[cat][sub]) Object.keys(arbolOperaciones[cat][sub]).forEach(d => el.innerHTML += `<option value="${d}">${d}</option>`);
            el.onchange = () => { poblarEmpOrigen(cat, sub, el.value); evaluarReglasExcel(); };
            limpiarNivelesDesde(4);
            evaluarReglasExcel();
        }
        function poblarEmpOrigen(cat, sub, det) {
            const el = document.getElementById('niv-emp-ori'); el.innerHTML = '<option value="">-- Seleccione Empresa Origen --</option>'; el.disabled = !det;
            if(det && arbolOperaciones[cat][sub][det]) Object.keys(arbolOperaciones[cat][sub][det]).forEach(e => el.innerHTML += `<option value="${e}">${e}</option>`);
            el.onchange = () => poblarCtaOrigen(cat, sub, det, el.value);
            limpiarNivelesDesde(5);
        }
        function poblarCtaOrigen(cat, sub, det, empOri) {
            const el = document.getElementById('niv-cta-ori'); el.innerHTML = '<option value="">-- Seleccione Cuenta Origen --</option>'; el.disabled = !empOri;
            if(empOri && arbolOperaciones[cat][sub][det][empOri]) Object.keys(arbolOperaciones[cat][sub][det][empOri]).forEach(c => el.innerHTML += `<option value="${c}">${c}</option>`);
            el.onchange = () => poblarEmpDestino(cat, sub, det, empOri, el.value);
            limpiarNivelesDesde(6);
        }
        function poblarEmpDestino(cat, sub, det, empOri, ctaOri) {

            if(ctaOri && arbolOperaciones[cat][sub][det][empOri][ctaOri]) Object.keys(arbolOperaciones[cat][sub][det][empOri][ctaOri]).forEach(e => el.innerHTML += `<option value="${e}">${e}</option>`);
            el.onchange = () => poblarCtaDestino(cat, sub, det, empOri, ctaOri, el.value);
            limpiarNivelesDesde(7);
        }
        function poblarCtaDestino(cat, sub, det, empOri, ctaOri, empDes) {
            const el = document.getElementById('niv-cta-des'); el.innerHTML = '<option value="">-- Seleccione Cuenta Destino --</option>'; el.disabled = !empDes;
            if(empDes && arbolOperaciones[cat][sub][det][empOri][ctaOri][empDes]) arbolOperaciones[cat][sub][det][empOri][ctaOri][empDes].forEach(c => el.innerHTML += `<option value="${c}">${c}</option>`);
        }
        function limpiarNivelesDesde(nv) {
            if(nv <= 3) { const e = document.getElementById('niv-det'); e.innerHTML='<option value=""></option>'; e.disabled=true; }
            if(nv <= 4) { const e = document.getElementById('niv-emp-ori'); e.innerHTML='<option value=""></option>'; e.disabled=true; }
            if(nv <= 5) { const e = document.getElementById('niv-cta-ori'); e.innerHTML='<option value=""></option>'; e.disabled=true; }
            if(nv <= 6) { const e = document.getElementById('niv-emp-des'); e.innerHTML='<option value=""></option>'; e.disabled=true; }
            if(nv <= 7) { const e = document.getElementById('niv-cta-des'); e.innerHTML='<option value=""></option>'; e.disabled=true; }
        }

        function cargarListaPlantillas() {
            const pls = JSON.parse(localStorage.getItem('bold_plantillas_bd') || '[]');
            const sel = document.getElementById('select-plantilla');
            sel.innerHTML = '<option value="">-- Seleccionar Plantilla Guardada --</option>';
            pls.forEach((p, i) => sel.innerHTML += `<option value="${i}">${p.nombre}</option>`);
        }

        function limpiarFormularioCompleto() {
            document.getElementById('nombre-op').value = "";
            document.getElementById('monto-sol').value = "";
            document.getElementById('doc-link').value = "";
            document.getElementById('doc-instrucciones').value = "";
            document.getElementById('doc-archivo').value = "";
            document.getElementById('prioridad-op').value = "2";
            ajustarANS();
            inicializarArbol();
            evaluarReglasExcel();
        }

        function cargarPlantilla() {
            const idx = document.getElementById('select-plantilla').value;
            if(idx === "") return alert("⚠️ Selecciona una plantilla del menú primero.");
            const p = JSON.parse(localStorage.getItem('bold_plantillas_bd') || '[]')[idx];
            if(!p) return;

            limpiarFormularioCompleto();
            document.getElementById('nombre-op').value = p.nombre;
            if(p.monto) document.getElementById('monto-sol').value = p.monto;
            if(p.moneda) document.getElementById('moneda-op').value = p.moneda;
            
            document.getElementById('niv-cat').value = p.cat;
            poblarSubtipos(p.cat);
            setTimeout(() => {
                document.getElementById('niv-sub').value = p.sub;
                poblarDetalles(p.cat, p.sub);
                setTimeout(() => {
                    document.getElementById('niv-det').value = p.det;
                    poblarEmpOrigen(p.cat, p.sub, p.det);
                    setTimeout(() => {
                        document.getElementById('niv-emp-ori').value = p.empOri;
                        poblarCtaOrigen(p.cat, p.sub, p.det, p.empOri);
                        setTimeout(() => {
                            document.getElementById('niv-cta-ori').value = p.ctaOri;
                            poblarEmpDestino(p.cat, p.sub, p.det, p.empOri, p.ctaOri);
                            setTimeout(() => {
                                document.getElementById('niv-emp-des').value = p.empDes;
                                poblarCtaDestino(p.cat, p.sub, p.det, p.empOri, p.ctaOri, p.empDes);
                                setTimeout(() => { document.getElementById('niv-cta-des').value = p.ctaDes; evaluarReglasExcel(); }, 40);
                            }, 40);
                        }, 40);
                    }, 40);
                }, 40);
            }, 40);
            alert(`⚡ Plantilla "${p.nombre}" cargada con éxito. Los campos anteriores han sido limpiados.`);
        }

        function guardarComoPlantilla() {
            const nom = document.getElementById('nombre-op').value.trim();
            if(!nom) return alert("⚠️ Es obligatorio escribir el 'Nombre de Operación / Plantilla' para poder guardar el modelo.");
            const cat = document.getElementById('niv-cat').value;
            const ctaDes = document.getElementById('niv-cta-des').value;
            if(!cat || !ctaDes) return alert("⚠️ Completa los 7 niveles del árbol para guardar la plantilla.");

            const pls = JSON.parse(localStorage.getItem('bold_plantillas_bd') || '[]');
            pls.push({
                nombre: nom,
                monto: document.getElementById('monto-sol').value,
                moneda: document.getElementById('moneda-op').value,
                cat: cat, sub: document.getElementById('niv-sub').value, det: document.getElementById('niv-det').value,
                empOri: document.getElementById('niv-emp-ori').value, ctaOri: document.getElementById('niv-cta-ori').value,
                empDes: document.getElementById('niv-emp-des').value, ctaDes: ctaDes
            });
            localStorage.setItem('bold_plantillas_bd', JSON.stringify(pls));
            cargarListaPlantillas();
            alert(`💾 ¡Modelo "${nom}" guardado exitosamente!\nAhora estará disponible en el menú superior para futuras cargas.`);
        }

        function procesarRadicacion() {
            const nomOp = document.getElementById('nombre-op').value.trim();
            const monto = document.getElementById('monto-sol').value;
            const ctaDes = document.getElementById('niv-cta-des').value;
            const ctaOri = document.getElementById('niv-cta-ori').value;
            const prio = document.getElementById('prioridad-op').value;
            const ans = document.getElementById('ans-op').value;
            const moneda = document.getElementById('moneda-op').value;
            const fileInput = document.getElementById('doc-archivo');
            const link = document.getElementById('doc-link').value.trim();
            const inst = document.getElementById('doc-instrucciones').value.trim();

            if(!nomOp) return alert("⚠️ El 'Nombre de Operación / Plantilla' es obligatorio.");
            if(!monto || monto <= 0) return alert("⚠️ Por favor ingresa un monto solicitado válido.");
            if(!ctaDes) return alert("⚠️ Por favor completa los 7 niveles de enrutamiento bancario (Cuenta Destino).");

            document.getElementById('conf-nom').innerText = nomOp;
            const prioTexto = prio === "1" ? "🔥 1 - Alta (Urgente)" : prio === "2" ? "⚡ 2 - Media (Normal)" : "☕ 3 - Baja (Flexible)";
            document.getElementById('conf-prio').innerText = `${prioTexto} | ANS: ${ans}`;
            document.getElementById('conf-monto').innerText = `$ ${parseFloat(monto).toLocaleString()} ${moneda}`;
            document.getElementById('conf-ori').innerText = ctaOri || "No seleccionada";
            document.getElementById('conf-des').innerText = ctaDes;

            let docResumen = [];
            if (fileInput && fileInput.files[0]) docResumen.push(`📁 Archivo: ${fileInput.files[0].name}`);
            if (link) docResumen.push("🔗 Link adjunto");
            if (inst) docResumen.push("📝 Con observaciones");
            document.getElementById('conf-doc').innerText = docResumen.length > 0 ? docResumen.join(" | ") : "Ninguno (Estándar)";

            document.getElementById('modalConfirmacion').style.display = 'flex';
        }

        function cerrarModalConfirmacion() {
            document.getElementById('modalConfirmacion').style.display = 'none';
        }

        function confirmarYRadicar() {
            cerrarModalConfirmacion();
            const fileInput = document.getElementById('doc-archivo');
            const file = fileInput ? fileInput.files[0] : null;

            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    finalizarGuardadoOperacion(e.target.result, file.name, file.type);
                };
                reader.readAsDataURL(file);
            } else {
                finalizarGuardadoOperacion(null, null, null);
            }
        }

        function finalizarGuardadoOperacion(fileData, fileName, fileType) {
            const ops = typeof obtenerOperaciones === 'function' ? obtenerOperaciones() : [];
            const usr = JSON.parse(sessionStorage.getItem('usuarioLogueado') || localStorage.getItem('bold_ultimo_usuario_backup') || '{}');
            const sec = String(ops.length + 1).padStart(4, '0');
            const rad = `TRAS-${new Date().toLocaleDateString('es-CO').substring(0,5)}-26-${sec}`;

            const nomOp = document.getElementById('nombre-op').value.trim();
            const cat = document.getElementById('niv-cat').value;
            const sub = document.getElementById('niv-sub').value;
            const det = document.getElementById('niv-det').value;
            const empOri = document.getElementById('niv-emp-ori').value;
            const ctaOri = document.getElementById('niv-cta-ori').value;
            const empDes = document.getElementById('niv-emp-des').value;
            const ctaDes = document.getElementById('niv-cta-des').value;
            const monto = document.getElementById('monto-sol').value;
            const prio = document.getElementById('prioridad-op').value;
            const ans = document.getElementById('ans-op').value;
            const moneda = document.getElementById('moneda-op').value;
            const link = document.getElementById('doc-link').value.trim();
            const inst = document.getElementById('doc-instrucciones').value.trim();

            const nueva = {
                radicado: rad,
                nombreOperacion: nomOp,
                pais: usr.pais || "Colombia",
                empresa: empOri,
                compDestino: empDes,
                tipo: `${cat} - ${sub}`,
                detalle: det,
                ctaOrigen: ctaOri,
                ctaDestino: ctaDes,
                solicitante: usr.correo || "lau@bold.co",
                montoSol: parseFloat(monto),
                montoPrep: parseFloat(monto),
                moneda: moneda,
                registros: 1,
                ans: ans,
                estado: "Pendiente Validación",
                prioridad: parseInt(prio),
                fechaRadicacion: typeof obtenerFechaHoraMilitar === 'function' ? obtenerFechaHoraMilitar().split(" - ")[0] : new Date().toLocaleDateString('es-CO'),
                fechaISO: typeof obtenerFechaISO === 'function' ? obtenerFechaISO() : "2026-07-27",
                apartadaPor: null,
                enTransito: true,
                comprobanteCerrado: false,
                archivoNombre: fileName || null,
                archivoTipo: fileType || null,
                archivoData: fileData || null,
                archivoLink: link || null,
                instrucciones: inst || null,
                historial: [{
                    fecha: typeof obtenerFechaHoraMilitar === 'function' ? obtenerFechaHoraMilitar() : "Hoy",
                    paso: "1. RADICACIÓN EN CASCADA",
                    detalle: `Operación "${nomOp}" radicada por ${usr.nombre || 'Solicitante'}. ${fileName ? '📎 Adjuntó archivo: '+fileName : ''}`
                }]
            };

            ops.unshift(nueva);
            if(typeof guardarOperaciones === 'function') guardarOperaciones(ops);
            if(typeof crearNotificacion === 'function') crearNotificacion(rad, `📝 ¡NUEVA RADICACIÓN! ${usr.nombre || 'Laura'} radicó: "${nomOp}" por $${parseFloat(monto).toLocaleString()} ${moneda}.`);
            
            alert(`✅ ¡Operación Radicada con Éxito!\nRadicado: ${rad}\nNombre: ${nomOp}\n\nSe ha notificado al equipo Preparador para su Validación Compliance y Montaje.`);
            limpiarFormularioCompleto();
            renderizarTabla();
        }
